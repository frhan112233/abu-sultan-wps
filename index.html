document.addEventListener('DOMContentLoaded', function () { const fileInput = document.getElementById('fileInput'); const processBtn = document.getElementById('processBtn'); const saveBtn = document.getElementById('saveBtn'); let processedContent = ""; // عند اختيار الملف fileInput.addEventListener('change', function(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { let text = e.target.result; // توحيد فواصل الأسطر وإجبارها لتكون CRLF مطابقة للملفات الناجحة text = text.replace(/\r\n/g, '\n').replace(/\n/g, '\r\n'); processedContent = text; }; // قراءة الملف بترميز UTF-8 لضمان سلامة النصوص reader.readAsText(file, 'UTF-8'); }); // زر معالجة وتجهيز الملفات (CRLF) processBtn.addEventListener('click', function() { if (!processedContent) { alert('الرجاء اختيار ملف نصي أولاً!'); return; } // تطبيق معالجة فواصل الأسطر للتأكيد عند الضغط processedContent = processedContent.replace(/\r\n/g, '\n').replace(/\n/g, '\r\n'); alert('تم تجهيز الملف ومعالجة الفواصل بنظام CRLF بنجاح!'); }); // زر حفظ الملف saveBtn.addEventListener('click', function() { if (!processedContent) { alert('لا توجد بيانات جاهزة للحفظ!'); return; } // إنشاء ملف Blob بترميز يدعم النظام وبصيغة TXT const blob = new Blob([processedContent], { type: 'text/plain;charset=utf-8' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'WPS_Processed_CRLF.txt'; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); }); });
