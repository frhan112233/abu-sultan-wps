document.getElementById('processBtn').addEventListener('click', function () { const fileInput = document.getElementById('fileInput'); const repeatCountInput = document.getElementById('repeatCount').value; const outputNameInput = document.getElementById('outputName').value; if (fileInput.files.length === 0) { alert('الرجاء اختيار ملف TXT أولاً'); return; } const file = fileInput.files[0]; const reader = new FileReader(); reader.onload = function (e) { let text = e.target.result; // توحيد فواصل الأسطر لضمان توافقها مع نظام Windows و مدد (CRLF) let lines = text.split(/\r?\n/); // هنا يتم معالجة وتكرار الأسطر حسب العدد المطلوب مع الحفاظ على البنية والتوقيع let repeatCount = parseInt(repeatCountInput) || 1; let processedLines = []; // نفترض أنك تريد تكرار أسطر البيانات بناءً على المنطق المطلوب لديك for (let i = 0; i < repeatCount; i++) { lines.forEach(line => { processedLines.push(line); }); } // إعادة تجميع الملف باستخدام فواصل الأسطر القياسية CRLF (\r\n) let finalContent = processedLines.join("\r\n"); // إنشاء ملف وتحميله تلقائياً let blob = new Blob([finalContent], { type: "text/plain;charset=utf-8" }); let link = document.createElement("a"); link.href = URL.createObjectURL(blob); link.download = (outputNameInput || "output") + ".txt"; link.click(); }; reader.readAsText(file); });
